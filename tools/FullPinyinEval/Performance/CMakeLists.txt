cmake_minimum_required(VERSION 3.20)
project(TigerClawPinyinWindowProbe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Explicit Windows real-window probe only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(OVERLAY "${CMAKE_CURRENT_SOURCE_DIR}/../../../next/TigerClaw.Overlay.Native" ABSOLUTE)
file(READ "${OVERLAY}/main.cpp" original)
set(present "try { renderer.Present(candidate_, &destination); }")
string(FIND "${original}" "${present}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "Renderer presentation path changed; update probe")
endif()
string(REPLACE "${present}" "try { renderer.Present(candidate_, &destination); latency_probe::Presented(state_, candidate_); }" instrumented "${original}")
set(show "frameForeground_ = GetForegroundWindow();\n            return true;")
string(FIND "${instrumented}" "${show}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "Candidate show path changed; update probe")
endif()
string(REPLACE "${show}" "frameForeground_ = GetForegroundWindow();\n            latency_probe::Presented(state_, candidate_);\n            return true;" instrumented "${instrumented}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/latency-overlay-main.cpp" "${instrumented}")
add_executable(pinyin_window_probe WindowProbe.cpp "${OVERLAY}/Renderer.cpp" "${OVERLAY}/Transport.cpp" "${OVERLAY}/Sound.cpp" "${OVERLAY}/Model.cpp")
target_include_directories(pinyin_window_probe PRIVATE "${OVERLAY}" "${CMAKE_CURRENT_BINARY_DIR}" "${OVERLAY}/../../third_party/llama.cpp/vendor")
target_compile_definitions(pinyin_window_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(pinyin_window_probe PRIVATE /utf-8 /W4)
target_link_libraries(pinyin_window_probe PRIVATE d2d1 dwrite user32 gdi32 shell32 shcore ole32 xaudio2 imm32 uuid dwmapi)

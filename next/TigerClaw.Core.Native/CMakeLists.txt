if(WIN32)
    # The installed runtime has exactly the same executable identity as C#.
    # Keep the existing isolated tool executable separate from production startup.
    add_executable(core_native_runtime main.cpp ReplayProbe.cpp LexiconTextProbe.cpp MainlineProbe.cpp RuntimeProtocolProbe.cpp)
    target_compile_definitions(core_native_runtime PRIVATE TIGER_CORE_PRODUCTION=1)
    target_link_libraries(core_native_runtime PRIVATE tiger_core_native)
    target_include_directories(core_native_runtime SYSTEM PRIVATE ../../third_party/llama.cpp/vendor)
    set_target_properties(core_native_runtime PROPERTIES OUTPUT_NAME "TigerClaw.Core")
    enable_language(RC)
    # Read the shared version without overwriting the user's BuildInfo.cs.
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../TigerClaw.Shared/BuildInfo.cs" CORE_BUILD_INFO)
    string(REGEX MATCH "VersionLabel = \"([^\"]+)\"" CORE_VERSION_MATCH "${CORE_BUILD_INFO}")
    set(TIGER_CORE_VERSION "${CMAKE_MATCH_1}")
    if(NOT TIGER_CORE_VERSION)
        set(TIGER_CORE_VERSION "1.0.0")
    endif()
    configure_file(CoreVersion.rc.in "${CMAKE_CURRENT_BINARY_DIR}/CoreVersion.rc" @ONLY)
    target_sources(core_native_runtime PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/CoreVersion.rc")
    if(MSVC)
        target_link_options(core_native_runtime PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/../app.manifest")
    endif()
endif()

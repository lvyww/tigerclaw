cmake_minimum_required(VERSION 3.20)
project(TigerClawOverlayNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(overlay_model Model.cpp)
target_include_directories(overlay_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(overlay_model SYSTEM PRIVATE ../../third_party/llama.cpp/vendor)
if(MSVC)
    target_compile_options(overlay_model PRIVATE /W4 /utf-8)
else()
    target_compile_options(overlay_model PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(WIN32)
    add_executable(overlay_native WIN32 main.cpp Renderer.cpp Transport.cpp Sound.cpp)
    target_include_directories(overlay_native SYSTEM PRIVATE ../../third_party/llama.cpp/vendor)
    target_compile_definitions(overlay_native PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(overlay_native PRIVATE overlay_model d2d1 dwrite user32 gdi32 shell32 shcore ole32)
    if(MSVC)
        target_link_libraries(overlay_native PRIVATE xaudio2)
    else()
        target_link_libraries(overlay_native PRIVATE xaudio2_9)
    endif()
    set_target_properties(overlay_native PROPERTIES OUTPUT_NAME "TigerClaw.Overlay")
    if(MINGW)
        target_link_options(overlay_native PRIVATE -municode -static)
    endif()
    if(MSVC)
        target_compile_options(overlay_native PRIVATE /W4 /utf-8)
    endif()
    add_custom_command(TARGET overlay_native POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/../TigerClaw.Overlay/sounds" "$<TARGET_FILE_DIR:overlay_native>/sounds"
        COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:overlay_native>"
        "$<TARGET_FILE_DIR:overlay_native>/TigerClaw.Overlay.Native.Preview.exe"
        COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/THIRD-PARTY-NOTICES.txt"
        "$<TARGET_FILE_DIR:overlay_native>/THIRD-PARTY-NOTICES.txt")
endif()
if(BUILD_TESTING)
    if(WIN32)
        add_executable(overlay_animation_tests tests/AnimationWakeTests.cpp Renderer.cpp)
        target_include_directories(overlay_animation_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_definitions(overlay_animation_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
        target_link_libraries(overlay_animation_tests PRIVATE overlay_model d2d1 dwrite user32 gdi32 ole32)
        add_executable(overlay_menu_tests tests/MenuTests.cpp)
        target_compile_definitions(overlay_menu_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
        target_link_libraries(overlay_menu_tests PRIVATE user32)
        add_dependencies(overlay_menu_tests overlay_native)
        # Explicit opt-in only: moves the pointer and clicks its own test window.
        add_executable(overlay_render_probe tests/RenderProbe.cpp Renderer.cpp)
        target_include_directories(overlay_render_probe SYSTEM PRIVATE ../../third_party/llama.cpp/vendor)
        target_compile_definitions(overlay_render_probe PRIVATE TIGERCLAW_RENDER_PROBE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
        target_link_libraries(overlay_render_probe PRIVATE overlay_model d2d1 dwrite user32 gdi32 ole32)
        # Hidden, test-owned layered HWND; explicitly invoked on Windows, no IPC.
        add_executable(overlay_sound_tests tests/SoundTests.cpp Sound.cpp)
        target_include_directories(overlay_sound_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_definitions(overlay_sound_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
        target_link_libraries(overlay_sound_tests PRIVATE ole32)
        if(MSVC)
            target_link_libraries(overlay_sound_tests PRIVATE xaudio2)
        else()
            target_link_libraries(overlay_sound_tests PRIVATE xaudio2_9)
            target_link_options(overlay_sound_tests PRIVATE -static)
        endif()
        add_dependencies(overlay_sound_tests overlay_native)
        # Requires an audio endpoint; run explicitly, without audible playback.
        add_executable(overlay_window_tests tests/WindowTests.cpp)
        target_include_directories(overlay_window_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_include_directories(overlay_window_tests SYSTEM PRIVATE ../../third_party/llama.cpp/vendor)
        target_compile_definitions(overlay_window_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
        target_link_libraries(overlay_window_tests PRIVATE user32)
        add_custom_target(overlay_test_copy ALL COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:overlay_native>" "$<TARGET_FILE_DIR:overlay_native>/TigerClaw.Overlay.Native.Test.exe"
            DEPENDS overlay_native)
        add_dependencies(overlay_window_tests overlay_test_copy)
        if(MINGW)
            target_link_options(overlay_window_tests PRIVATE -static)
        endif()
        # Explicit opt-in execution: this test displays real isolated windows.
        # Do not add it to unattended CTest or the non-GUI PowerShell runner.
        add_executable(overlay_transport_tests tests/TransportTests.cpp Transport.cpp)
        target_include_directories(overlay_transport_tests SYSTEM PRIVATE ../../third_party/llama.cpp/vendor)
        target_compile_definitions(overlay_transport_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
        target_link_libraries(overlay_transport_tests PRIVATE overlay_model user32)
        if(MINGW)
            target_link_options(overlay_transport_tests PRIVATE -static)
        endif()
        add_test(NAME overlay_transport COMMAND overlay_transport_tests)
        set_tests_properties(overlay_transport PROPERTIES TIMEOUT 45)
        add_custom_command(TARGET overlay_transport_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
            "${CMAKE_CURRENT_SOURCE_DIR}/Run-IsolatedTests.ps1" "$<TARGET_FILE_DIR:overlay_transport_tests>/Run-IsolatedTests.ps1")
    endif()
    add_executable(overlay_format_probe tests/FormatProbe.cpp)
    target_link_libraries(overlay_format_probe PRIVATE overlay_model)
    target_include_directories(overlay_format_probe SYSTEM PRIVATE ../../third_party/llama.cpp/vendor)
    add_executable(overlay_model_tests tests/ModelTests.cpp)
    target_link_libraries(overlay_model_tests PRIVATE overlay_model)
    if(MSVC)
        target_compile_options(overlay_model_tests PRIVATE /utf-8)
    endif()
    if(MINGW)
        target_link_options(overlay_model_tests PRIVATE -static)
        target_link_options(overlay_format_probe PRIVATE -static)
    endif()
    add_test(NAME overlay_model COMMAND overlay_model_tests)
    include(HistoryTests.cmake)
    include(PendingFrameTests.cmake)
endif()

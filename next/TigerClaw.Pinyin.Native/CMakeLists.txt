cmake_minimum_required(VERSION 3.20)
project(TigerClawPinyinNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(KENLM "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/kenlm")
# Query-only build: no Boost, trainer, command-line programs or compressed ARPA.
file(GLOB SOURCES "${KENLM}/lm/*.cc" "${KENLM}/util/*.cc" "${KENLM}/util/double-conversion/*.cc")
list(FILTER SOURCES EXCLUDE REGEX "(_test|_main)\\.cc$")
add_library(jointkenlm SHARED kenlm_bridge.cpp ${SOURCES})
target_include_directories(jointkenlm PRIVATE "${KENLM}")
target_compile_definitions(jointkenlm PRIVATE KENLM_MAX_ORDER=6)
set_target_properties(jointkenlm PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  set_property(TARGET jointkenlm PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_definitions(jointkenlm PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(jointkenlm PRIVATE /utf-8)
  target_link_libraries(jointkenlm PRIVATE psapi)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(jointkenlm PRIVATE Threads::Threads)
endif()
